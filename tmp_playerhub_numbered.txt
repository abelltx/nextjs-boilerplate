  58: function detectPromptTarget(prompt: string): PromptTarget {
  59:   const lower = String(prompt ?? "").toLowerCase();
  60:   if (!lower.trim()) return null;
  61: 
  62:   const dieMatch = lower.match(/\bd\s*(4|6|8|10|12|20|100)\b/);
  63:   if (dieMatch) return { kind: "die", die: Number(dieMatch[1]) };
  64: 
  65:   if (/\bstrength\b|\bstr\b/.test(lower)) return { kind: "ability", abilityKey: "str" };
  66:   if (/\bdexterity\b|\bdex\b/.test(lower)) return { kind: "ability", abilityKey: "dex" };
  67:   if (/\bconstitution\b|\bcon\b/.test(lower)) return { kind: "ability", abilityKey: "con" };
  68:   if (/\bintelligence\b|\bint\b/.test(lower)) return { kind: "ability", abilityKey: "int" };
  69:   if (/\bwisdom\b|\bwis\b/.test(lower)) return { kind: "ability", abilityKey: "wis" };
  70:   if (/\bcharisma\b|\bcha\b/.test(lower)) return { kind: "ability", abilityKey: "cha" };
  71: 
  72:   for (const skill of SKILL_ALIASES) {
  73:     for (const alias of skill.aliases) {
  74:       if (lower.includes(alias)) return { kind: "skill", skillKey: skill.key };
  75:     }
  76:   }
  77: 
  78:   return null;
  79: }

  81: function isLiveState(state: any) {
  82:   if (!state) return false;
  83:   if (state.player_view === true) return true;
  84:   if (state.is_live === true) return true;
  85:   if (state.live === true) return true;
  86:   if (state.roll_open === true) return true;
  87:   if (typeof state.presented_block_id === "string" && state.presented_block_id.trim().length > 0) return true;
  88:   return false;
  89: }

 721: function ActionListPanel(props: {
 722:   actions: Array<{
 723:     id: string;
 724:     name: string;
 725:     type?: string | null;
 726:     summary?: string | null;
 727:     rules_text?: string | null;
 728:     range_normal?: number | null;
 729:     range_max?: number | null;
 730:     uses_attack_roll?: boolean | null;
 731:     save_ability?: string | null;
 732:     save_dc_override?: number | null;
 733:     damage_dice?: string | null;
 734:     damage_bonus?: number | null;
 735:     attack_bonus_override?: number | null;
 736:     damage_type?: string | null;
 737:     on_fail?: string | null;
 738:     on_success?: string | null;
 739:   }>;
 740: }) {
 741:   const [rolls, setRolls] = useState<Record<string, { hit?: string; damage?: string }>>({});
 742:   const HEAL_TYPES = new Set(["healing", "temporary_hp"]);
 743: 
 744:   function rollDie(sides: number) {
 745:     return Math.floor(Math.random() * sides) + 1;
 746:   }
 747: 
 748:   function rollHit(action: any) {
 749:     const bonus = Number(action.attack_bonus_override ?? 0);
 750:     const d20 = rollDie(20);
 751:     const total = d20 + (Number.isFinite(bonus) ? bonus : 0);
 752:     setRolls((prev) => ({
 753:       ...prev,
 754:       [action.id]: {
 755:         ...(prev[action.id] ?? {}),
 756:         hit: `${total} (d20 ${d20}${bonus ? ` + ${bonus}` : ""})`,
 757:       },
 758:     }));
 759:   }
 760: 
 761:   function rollDamage(action: any) {
 762:     const formula = String(action.damage_dice ?? "").trim().toLowerCase();
 763:     const bonusFromField = Number(action.damage_bonus ?? 0);
 764:     const m = formula.match(/^(\d*)d(\d+)([+-]\d+)?$/i);
 765:     if (!m) return;
 766:     const count = Math.max(1, Number(m[1] || 1));
 767:     const sides = Math.max(2, Number(m[2] || 2));
 768:     const inlineBonus = Number(m[3] || 0);
 769:     const bonus = (Number.isFinite(inlineBonus) ? inlineBonus : 0) + (Number.isFinite(bonusFromField) ? bonusFromField : 0);
 770:     const rollsArr = Array.from({ length: count }, () => rollDie(sides));
 771:     const total = rollsArr.reduce((t, n) => t + n, 0) + bonus;
 772:     const outcomeLabel = HEAL_TYPES.has(String(action.damage_type ?? "").toLowerCase()) ? "heal" : "damage";
 773:     setRolls((prev) => ({
 774:       ...prev,
 775:       [action.id]: {
 776:         ...(prev[action.id] ?? {}),
 777:         damage: `${total} ${outcomeLabel} ([${rollsArr.join(", ")}]${bonus ? ` ${bonus > 0 ? "+" : "-"} ${Math.abs(bonus)}` : ""})`,
 778:       },
 779:     }));
 780:   }
 781: 
 782:   return (
 783:     <div className="rounded-2xl border border-neutral-800 bg-neutral-950/40 p-4">
 784:       <div className="text-sm font-semibold">Actions</div>
 785:       {props.actions.length ? (
 786:         <div className="mt-3 overflow-hidden rounded-xl border border-neutral-800">
 787:           <div className="grid grid-cols-12 gap-2 border-b border-neutral-800 bg-neutral-900/50 px-3 py-2 text-[11px] uppercase tracking-wide text-neutral-400">
 788:             <div className="col-span-3">Attack</div>
 789:             <div className="col-span-2">Range</div>
 790:             <div className="col-span-2">Hit / DC</div>
 791:             <div className="col-span-2">Damage</div>
 792:             <div className="col-span-3">Notes</div>
 793:           </div>
 794:           {props.actions.map((a) => (
 795:             <div key={a.id} className="grid grid-cols-12 gap-2 border-b border-neutral-800 px-3 py-2 text-sm last:border-b-0">
 796:               <div className="col-span-3 min-w-0">
 797:                 <div className="truncate font-semibold text-neutral-100">{a.name}</div>
 798:                 <div className="text-[11px] text-neutral-400">{a.type ?? "other"}</div>
 799:               </div>
 800:               <div className="col-span-2 text-xs text-neutral-300">
 801:                 {a.range_normal ? `${a.range_normal} ft.` : ""}
 802:                 {a.range_max ? ` (${a.range_max})` : ""}
 803:               </div>
 804:               <div className="col-span-2 text-xs text-neutral-300">
 805:                 {a.uses_attack_roll !== false && a.attack_bonus_override != null ? (
 806:                   <div className="space-y-1">
 807:                     <div>+{a.attack_bonus_override}</div>
 808:                     <button
 809:                       type="button"
 810:                       className="rounded border border-neutral-700 px-2 py-0.5 text-[11px] hover:bg-neutral-900"
 811:                       onClick={() => rollHit(a)}
 812:                     >
 813:                       Roll Hit
 814:                     </button>
 815:                     {rolls[a.id]?.hit ? <div className="text-emerald-300">{rolls[a.id]?.hit}</div> : null}
 816:                   </div>
 817:                 ) : a.save_dc_override != null ? (
 818:                   <div>{`DC ${a.save_dc_override}${a.save_ability ? ` ${String(a.save_ability).toUpperCase()}` : ""}`}</div>
 819:                 ) : null}
 820:               </div>
 821:               <div className="col-span-2 text-xs text-neutral-300">
 822:                 {a.damage_dice ? (
 823:                   <div>
 824:                     <div>{`${a.damage_dice}${a.damage_type ? ` ${a.damage_type}` : ""}`}</div>
 825:                     <button
 826:                       type="button"
 827:                       className="mt-1 rounded border border-neutral-700 px-2 py-0.5 text-[11px] hover:bg-neutral-900"
 828:                       onClick={() => rollDamage(a)}
 829:                     >
 830:                       {HEAL_TYPES.has(String(a.damage_type ?? "").toLowerCase()) ? "Roll Heal" : "Roll Dmg"}
 831:                     </button>
 832:                     {rolls[a.id]?.damage ? <div className="text-emerald-300">{rolls[a.id]?.damage}</div> : null}
 833:                   </div>
 834:                 ) : null}
 835:               </div>
 836:               <div className="col-span-3 text-xs text-neutral-300">
 837:                 {a.summary?.trim() ? <div>{a.summary}</div> : null}
 838:                 {!a.summary?.trim() && a.rules_text?.trim() ? (
 839:                   <div className="line-clamp-2">{a.rules_text}</div>
 840:                 ) : null}
 841:                 {a.on_fail?.trim() ? <div>{`On fail: ${a.on_fail}`}</div> : null}
 842:                 {a.on_success?.trim() ? <div>{`On success: ${a.on_success}`}</div> : null}
 843:               </div>

 850:     </div>
